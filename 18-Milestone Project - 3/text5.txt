def build_alias_dict() -> dict[str, str]:
    """Build and return the merged signal alias dictionary.

    Merges ``DEFAULT_ALIASES`` with entries from ``aliases.json`` if the file
    exists and is valid. Entries in ``aliases.json`` overwrite defaults.

    Returns:
        Mapping of model signal names to their measurement signal equivalents.

    Note:
        If ``aliases.json`` is missing, malformed, or unreadable, the function
        logs the issue and returns the default aliases unchanged.
    """
    alias: dict[str, str] = dict(DEFAULT_ALIASES)

    app_dir = Path(__file__).resolve().parent
    aliases_json = app_dir / "aliases.json"

    if not aliases_json.exists():
        logger.debug("No aliases.json found at %s – using DEFAULT_ALIASES only", aliases_json)
        return alias

    try:
        text = aliases_json.read_text(encoding="utf-8")
        data = json.loads(text)

        if not isinstance(data, dict):
            logger.warning(
                "aliases.json ignored: expected JSON object (dict), got %s",
                type(data).__name__,
            )
            return alias

        normalized = {str(k): str(v) for k, v in data.items()}
        alias.update(normalized)

        logger.info(
            "Loaded %d external aliases from %s",
            len(normalized),
            aliases_json,
        )

    except json.JSONDecodeError as ex:
        logger.error(
            "Failed to parse aliases.json (%s): %s",
            aliases_json,
            ex,
        )

    except OSError as ex:
        logger.error(
            "Failed to read aliases.json (%s): %s",
            aliases_json,
            ex,
        )

    except Exception:
        logger.exception(
            "Unexpected error while loading aliases.json (%s)",
            aliases_json,
        )

    return alias
